### Step 1 — Paddle.gd: add velocity tracking

Open `scenes/game/Paddle.gd`.

**Add to the `# ── State ──` section**:
```gdscript
# Public — read by Ball.gd during paddle contact to compute spin.
# Updated each physics frame as the delta-position per second.
# [JS equiv]  get paddleVelocityX() { return this._paddleVelocityX }
var paddle_velocity_x: float = 0.0

# Previous frame's X position — used to compute velocity.
# [JS equiv]  #prevX = 0.0  (private class field)
var _prev_x: float = 0.0
```

**Add `_physics_process` method** (or insert into existing one if present):
```gdscript
# Track horizontal velocity each physics frame.
# Called at 60 Hz — same rate as Ball.gd's _physics_process.
# [JS equiv]  // called in requestAnimationFrame loop
func _physics_process(delta: float) -> void:
	# Velocity = displacement / time. Capped to prevent division issues if delta spikes.
	# [JS equiv]  this._paddleVelocityX = (this.x - this._prevX) / delta
	if delta > 0.0:
		paddle_velocity_x = (position.x - _prev_x) / delta
	_prev_x = position.x
```

> **Note**: `AnimatableBody2D` does NOT have a built-in velocity property. This manual tracking is the correct approach.

### Step 2 — Ball.gd: apply spin at paddle contact

Open `scenes/game/Ball.gd`. Find the `AnimatableBody2D` branch in `_move_and_reflect()`:

```gdscript
if collider is AnimatableBody2D:
    var paddle_half: float = collider.get_node("CollisionShape2D").shape.size.x * 0.5
    var contact_x: float = clampf(
        (global_position.x - collider.global_position.x) / paddle_half,
        -1.0, 1.0
    )
    var angle: float = contact_x * (PI / 3.0)
    _velocity = Vector2(sin(angle), -abs(cos(angle))) * _current_speed
```

**Add spin constant** in the constants section:
```gdscript
# Maximum spin added by paddle horizontal velocity at impact.
# ±20° in radians. Capped so the ball never flies nearly horizontal.
# [JS equiv]  const MAX_SPIN_RAD = 0.349  (20 * Math.PI / 180)
const MAX_SPIN_RAD: float = 0.349
# Maximum absolute angle from vertical (70°) — safety ceiling for combined angle + spin.
# [JS equiv]  const MAX_ANGLE_FROM_VERTICAL = 1.222  (70 * Math.PI / 180)
const MAX_ANGLE_FROM_VERTICAL: float = 1.222
```

**Replace the angle calculation block**:
```gdscript
if collider is AnimatableBody2D:
    var paddle_half: float = collider.get_node("CollisionShape2D").shape.size.x * 0.5
    var contact_x: float = clampf(
        (global_position.x - collider.global_position.x) / paddle_half,
        -1.0, 1.0
    )
    # Base angle from contact position: ±60° range (PI/3 = 60°).
    var angle: float = contact_x * (PI / 3.0)

    # Spin: paddle velocity at impact shifts the angle further in the paddle's travel direction.
    # Normalise paddle speed against a reference speed (600 px/s) — above that, full spin.
    # clampf() = Math.max(min, Math.min(max, v)) in JS.
    var spin: float = clampf(collider.paddle_velocity_x / 600.0, -1.0, 1.0) * MAX_SPIN_RAD
    # [JS equiv]  const finalAngle = clamp(angle + spin, -MAX_ANGLE, MAX_ANGLE)
    angle = clampf(angle + spin, -MAX_ANGLE_FROM_VERTICAL, MAX_ANGLE_FROM_VERTICAL)

    # Build velocity: x = sin(angle), y = -|cos(angle)| (always upward).
    # [JS equiv]  velocity = { x: Math.sin(angle), y: -Math.abs(Math.cos(angle)) } * speed
    _velocity = Vector2(sin(angle), -abs(cos(angle))) * _current_speed
```
