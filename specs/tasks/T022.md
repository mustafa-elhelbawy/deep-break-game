**Step 2 — Write `scenes/ui/HUD.gd`**

```gdscript
# HUD.gd
# Displays score and lives. Connects to GameManager signals on _ready().
# Lives displayed as "♥ N" text; Phase 4 replaces with icon sprites.
#
# [GDScript] 'extends CanvasLayer' — renders above all Node2D content.
# [JS equiv]  class HUD { constructor() { this._score = 0; this._lives = 3; } }
extends CanvasLayer

# ── Cached node references ─────────────────────────────────────────────────────
# [GDScript] @onready resolves paths after the scene tree is built.
# [JS equiv]  this._scoreLabel = document.getElementById('score')
@onready var _score_label: Label = $ScoreLabel
@onready var _lives_label: Label = $LivesLabel

# ── Lifecycle ──────────────────────────────────────────────────────────────────

func _ready() -> void:
	# Connect to GameManager signals so HUD updates automatically.
	# [GDScript] signal.connect(Callable) — registers a listener.
	# [JS equiv]  gameManager.addEventListener('scoreChanged', this.updateScore.bind(this))
	GameManager.score_changed.connect(update_score)
	GameManager.lives_changed.connect(update_lives)

	# Initialise display to current GameManager state.
	update_score(GameManager.score)
	update_lives(GameManager.lives)

# ── Public API (called directly by Game.gd and via signal) ────────────────────

## Update the score display.
## [JS equiv]  updateScore(newScore) { this._scoreLabel.textContent = String(newScore) }
func update_score(new_score: int) -> void:
	_score_label.text = str(new_score)

## Update the lives display.
## [JS equiv]  updateLives(newLives) { this._livesLabel.textContent = `♥ ${newLives}` }
func update_lives(new_lives: int) -> void:
	_lives_label.text = "♥ %d" % new_lives
```
