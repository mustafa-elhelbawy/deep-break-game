### Step 2 — Write PowerUp.gd

Attach a new script at `scenes/game/PowerUp.gd`:

```gdscript
# PowerUp.gd
# A falling token that drifts downward after a brick destroy.
# Area2D detects contact with the paddle (AnimatableBody2D on layer 1).
# Emits 'caught(type)' when the paddle overlaps it.
# Auto-removes when it exits the bottom of the screen.
#
# [GDScript] extends Area2D — no physics body, purely an overlap sensor.
# [JS equiv]  class PowerUpToken extends GameObject { ... }
extends Area2D

# ── Signals ────────────────────────────────────────────────────────────────────
# [GDScript] typed signal — Game.gd connects to this.
# [JS equiv]  this.addEventListener('caught', callback)
signal caught(type: String)

# ── Constants ──────────────────────────────────────────────────────────────────
# Fall speed in pixels per second. Slow enough to give the player time to react.
# [JS equiv]  const FALL_SPEED = 200
const FALL_SPEED: float = 200.0

# Type-to-colour map — visual identity for each power-up.
# [GDScript] Dictionary literal — like a plain JS object { W: '#00c8b8', ... }
const TYPE_COLOURS: Dictionary = {
	"W": Color("#00c8b8"),   # teal  — Wide Paddle
	"S": Color("#00e0ff"),   # cyan  — Slow Motion
	"M": Color("#ffc107"),   # amber — Multi-Ball
}

# ── State ──────────────────────────────────────────────────────────────────────
# [GDScript] typed member variable — set once by Brick.gd before adding to scene tree.
var _type: String = ""
var _screen_height: float = 780.0

# ── Setup ──────────────────────────────────────────────────────────────────────

## Called by Brick.gd immediately after instantiation, before add_child().
## type: "W", "S", or "M"
func setup(type: String) -> void:
	_type = type
	# Set icon colour based on type. Falls back to white if type is unknown.
	# [GDScript] get() returns the value or the default if key not found.
	# [JS equiv]  this._icon.style.background = TYPE_COLOURS[type] ?? 'white'
	var colour: Color = TYPE_COLOURS.get(type, Color.WHITE)
	$Icon.color = colour

func _ready() -> void:
	_screen_height = get_viewport().get_visible_rect().size.y
	# Connect body_entered — fires when a physics body enters this Area2D.
	# The paddle (AnimatableBody2D) is a physics body, so it triggers this.
	# [JS equiv]  this.addEventListener('collisionEnter', handler)
	body_entered.connect(_on_body_entered)

# ── Physics ────────────────────────────────────────────────────────────────────

func _physics_process(delta: float) -> void:
	# Move downward each frame.
	# In Godot 2D, +Y is downward.
	# [JS equiv]  this.y += FALL_SPEED * delta
	position.y += FALL_SPEED * delta

	# Auto-remove if off the bottom of the screen.
	# [JS equiv]  if (this.y > screenHeight + 40) this.remove()
	if position.y > _screen_height + 40.0:
		queue_free()

# ── Catch detection ────────────────────────────────────────────────────────────

func _on_body_entered(body: Node) -> void:
	# Only catch when the body is the paddle (AnimatableBody2D).
	# [GDScript] 'is' = instanceof check. AnimatableBody2D is the paddle type.
	# [JS equiv]  if (body instanceof Paddle)
	if body is AnimatableBody2D:
		# Emit caught signal with the power-up type string.
		# Game.gd is connected to this signal and will apply the effect.
		# [JS equiv]  this.dispatchEvent(new CustomEvent('caught', { detail: this._type }))
		emit_signal("caught", _type)
		# Remove immediately — no visual linger after catch.
		queue_free()
```
