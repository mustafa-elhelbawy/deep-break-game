Replace with:

**Add `_on_ball_lost()` handler**:
```gdscript
## Called whenever ANY active ball (primary or extra) emits the 'lost' signal.
func _on_ball_lost() -> void:
	_ball_count -= 1
	# [JS equiv]  if (this._ballCount <= 0) gameManager.loseLife()
	if _ball_count <= 0:
		# All balls lost — standard life-lost flow.
		_ball_count = 1  # reset for when play resumes
		GameManager.lose_life()
	# If _ball_count > 0: a secondary ball was lost, game continues silently.
```

### Step 2 — Add _activate_multiball()

```gdscript
func _activate_multiball() -> void:
	# Only spawn one extra ball at a time — catching M while active does nothing.
	# This avoids exponential ball accumulation.
	# [JS equiv]  if (this._extraBalls.length > 0) return
	if _extra_balls.size() > 0:
		return

	# Instantiate a second ball using the same packed scene as the primary ball.
	# [JS equiv]  const extraBall = new BallScene()
	var extra: CharacterBody2D = BallScene.instantiate()
	add_child(extra)

	# Position it slightly offset from the primary ball to avoid instant collision.
	extra.global_position = _ball.global_position + Vector2(20.0, 0.0)

	# Apply current level speed and any active slow modifier.
	extra.set_speed_for_level(GameManager.level)
	if _slow_timer != null and is_instance_valid(_slow_timer):
		extra.activate_slow()

	# Launch in slightly different direction for variety.
	# [JS equiv]  extraBall.launch({ x: 0.3, y: -1.0 })  (mirror angle of primary)
	extra.launch(Vector2(0.3, -1.0))

	# Track it and wire its lost signal.
	_extra_balls.append(extra)
	_ball_count += 1
	# [JS equiv]  extraBall.addEventListener('lost', () => this._onExtraBallLost(extraBall))
	extra.lost.connect(func(): _on_extra_ball_lost(extra))

func _on_extra_ball_lost(ball: CharacterBody2D) -> void:
	# Remove from tracking list.
	_extra_balls.erase(ball)
	# Trigger the shared counter logic.
	_on_ball_lost()
```

```gdscript
func _clear_extra_balls() -> void:
```
