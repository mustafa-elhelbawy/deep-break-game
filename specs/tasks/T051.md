### Step 2 — Add power-up type to brick descriptors

**Add a pool constant** near the top of the file (after `DIFFICULTY_TABLE`):

```gdscript
# Power-up types available per level tier.
# Each sub-array lists the types that CAN appear at or above that level threshold.
# [GDScript] Array of Dictionaries — like JS: [{ threshold: 5, types: ['W'] }, ...]
const POWERUP_TABLE: Array = [
	{ "threshold": 5,  "types": ["W"] },
	{ "threshold": 10, "types": ["W", "S"] },
	{ "threshold": 20, "types": ["W", "S", "M"] },
]

# Chance that an eligible brick carries a power-up (0.12 = 12%).
# [JS equiv]  const POWERUP_CHANCE = 0.12
const POWERUP_CHANCE: float = 0.12
```

**Add a helper to get available types for a level**:
```gdscript
## Returns the array of power-up type strings available at the given level.
## Returns an empty array if the level is below the first threshold.
## [JS equiv]  _getAvailablePowerups(level) { return table.filter(...).at(-1)?.types ?? [] }
func _get_available_powerups(level: int) -> Array:
	var pool: Array = []
	for entry in POWERUP_TABLE:
		if level >= entry["threshold"]:
			pool = entry["types"]
	return pool
```

**Inside the brick descriptor loop** (after the `tier` assignment, before `result.append()`), add power-up assignment:

```gdscript
			# Determine power-up type for this brick.
			# Only bricks with hp > 0 (destructible) can carry power-ups.
			# Indestructible bricks (hp == 0) do not drop anything.
			# [JS equiv]  const puType = hp > 0 && rng.random() < POWERUP_CHANCE ? pickRandom(pool) : ''
			var powerup_type: String = ""
			if hp > 0:
				var pool: Array = _get_available_powerups(level)
				if pool.size() > 0 and rng.randf() < POWERUP_CHANCE:
					# Pick a random type from the available pool.
					# randi_range(0, n-1) returns a random integer in [0, n-1].
					# [JS equiv]  pool[Math.floor(rng.random() * pool.length)]
					powerup_type = pool[rng.randi_range(0, pool.size() - 1)]

			result.append({
				"grid_pos":     Vector2i(col, row),
				"hit_points":   hp,
				"tier":         tier,
				"powerup_type": powerup_type,   # "" = none; "W"/"S"/"M" = power-up
			})
```

> **Important**: the two `rng` calls for `powerup_type` (`randf()` and `randi_range()`) must be placed AFTER the existing `rng` calls for `hp` and `indestructible_chance`. Adding calls anywhere else would shift all subsequent random draws and break seed reproducibility for the existing grid layout.
