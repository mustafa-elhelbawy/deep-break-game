### Step 1 — BrickGrid.gd: expose mobile bricks

Open `scenes/game/BrickGrid.gd`. Add a method to return all currently mobile bricks:

```gdscript
## Returns all brick children that are currently in mobile mode.
## Used by Game.gd to inject paddle_y and connect the reached_paddle signal.
## [JS equiv]  getMobileBricks() { return this._bricks.filter(b => b.isMobile) }
func get_mobile_bricks() -> Array:
	var result: Array = []
	for child in get_children():
		# Check if the brick has the _is_mobile flag set (defined in T057).
		if child.has_method("set_paddle_y") and child._is_mobile:
			result.append(child)
	return result
```

### Step 2 — Game.gd: wire mobile bricks after spawn

Open `scenes/game/Game.gd`. In `_spawn_level()`, after `_grid.spawn(descriptors)`, add:

```gdscript
func _spawn_level() -> void:
	var level_data: Dictionary = LevelGenerator.generate(GameManager.seed, GameManager.level)
	var descriptors: Array = level_data["bricks"]
	_grid.spawn(descriptors)

	# Wire mobile bricks — inject paddle Y and connect reached_paddle signal.
	_wire_mobile_bricks()

func _wire_mobile_bricks() -> void:
	# The paddle Y threshold is the paddle's current world-space Y position.
	# Mobile bricks emit reached_paddle when they cross this Y.
	# [JS equiv]  const paddleY = this._paddle.globalPosition.y
	var paddle_y: float = _paddle.global_position.y

	for brick in _grid.get_mobile_bricks():
		# Override the placeholder Y set in BrickGrid.gd (T058).
		brick.set_paddle_y(paddle_y)
		# Connect reached_paddle → lose life for this brick.
		# Using a lambda captures 'brick' so we can also shake + flash on the hit.
		# [JS equiv]  brick.addEventListener('reachedPaddle', () => this._onMobileBrickHit())
		if not brick.reached_paddle.is_connected(_on_mobile_brick_hit):
			brick.reached_paddle.connect(_on_mobile_brick_hit)

func _on_mobile_brick_hit() -> void:
	# A mobile brick reached the paddle — treat as a lost-ball event.
	# Visual feedback: dark red flash and shake (same as ball lost).
	JuiceManager.flash(Color("#330000"), 0.3)
	JuiceManager.shake(0.5, 0.15)
	# Lose a life through the standard game-manager flow.
	# [JS equiv]  gameManager.loseLife()
	GameManager.lose_life()
```
