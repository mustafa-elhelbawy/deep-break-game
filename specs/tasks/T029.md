**Full updated constants + state section** (replace the existing block):

```gdscript
# ── Constants ──────────────────────────────────────────────────────────────────
# BASE_SPEED is the reference value at level 1 — never mutated.
# _current_speed is the actual speed used this session, scaled by set_speed_for_level().
# [GDScript] 'const' = immutable; 'var' = mutable member variable.
# [JS equiv]  const BASE_SPEED = 400;  let _currentSpeed = 400;
const BASE_SPEED: float = 400.0

# Minimum absolute vertical speed enforced after every bounce.
# Prevents near-horizontal ball travel regardless of current speed.
# [JS equiv]  const MIN_VERTICAL_SPEED = 120
const MIN_VERTICAL_SPEED: float = 120.0

# Stuck-ball watchdog backup (per-bounce clamp handles the common case).
const STUCK_TIMEOUT: float = 2.0

# ── State ──────────────────────────────────────────────────────────────────────
var _velocity: Vector2    = Vector2.ZERO
var _active: bool         = false
var _stuck_timer: float   = 0.0
var _current_speed: float = BASE_SPEED  # scaled by set_speed_for_level() before each serve
```

**New public method — add after `deactivate()`**:

```gdscript
## Sets ball speed for the given level number. Call this BEFORE launch() on each serve.
## Speed increases 8% per level, capped at 2× base speed (800 px/s at level 13+).
## [JS equiv]  setSpeedForLevel(level) { this._currentSpeed = Math.min(1 + level * 0.08, 2) * BASE_SPEED }
func set_speed_for_level(level: int) -> void:
	# minf() is the float version of min() — same as Math.min() in JS.
	var scale: float = minf(1.0 + level * 0.08, 2.0)
	_current_speed = BASE_SPEED * scale
```
