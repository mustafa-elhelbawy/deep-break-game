### Step 1 — Paddle.gd: add wide paddle methods

Open `scenes/game/Paddle.gd`.

**Add constants**:
```gdscript
# Wide paddle scale factor and duration.
const WIDE_SCALE: float    = 1.8
const WIDE_DURATION: float = 10.0
```

**Add state variables**:
```gdscript
# Reference to the normal (un-scaled) paddle width — set once in _ready().
# Used to revert after wide effect expires.
# [JS equiv]  #normalWidth = 120
var _normal_width: float = 120.0
var _wide_timer: SceneTreeTimer = null   # null when not active
```

**Update `_ready()`** to store the normal width:
```gdscript
func _ready() -> void:
	var viewport_rect: Rect2 = get_viewport().get_visible_rect()
	_screen_width  = viewport_rect.size.x
	_screen_height = viewport_rect.size.y
	# Store normal width for revert.
	_normal_width = ($CollisionShape2D.shape as RectangleShape2D).size.x
	_half_width   = _normal_width / 2.0
```

**Add activate and revert methods**:
```gdscript
## Activates the Wide Paddle effect. Safe to call while already active — resets timer.
func activate_wide() -> void:
	# Apply 1.8× scale to collision shape width and sprite.
	var new_width: float = _normal_width * WIDE_SCALE
	# [GDScript] Cast shape to RectangleShape2D to access size property.
	# [JS equiv]  this.collisionShape.size.x = newWidth
	($CollisionShape2D.shape as RectangleShape2D).size.x = new_width
	$Sprite2D.scale.x = WIDE_SCALE
	# Update half_width used by _move_to_x() clamping.
	_half_width = new_width / 2.0

	# Cancel any existing timer before starting a new one.
	# This is the "replacement" behaviour — catching W again resets the 10 s.
	if _wide_timer != null and is_instance_valid(_wide_timer):
		# SceneTreeTimers can't be cancelled directly — but we track the reference.
		# Disconnect old timeout signal to prevent double revert.
		if _wide_timer.timeout.is_connected(_revert_wide):
			_wide_timer.timeout.disconnect(_revert_wide)

	# Start a new 10-second timer.
	# [GDScript] create_timer() returns a SceneTreeTimer — fire-and-forget.
	# [JS equiv]  this._wideTimer = setTimeout(() => this._revertWide(), 10000)
	_wide_timer = get_tree().create_timer(WIDE_DURATION)
	_wide_timer.timeout.connect(_revert_wide)

func _revert_wide() -> void:
	# Restore collision shape and sprite to normal dimensions.
	($CollisionShape2D.shape as RectangleShape2D).size.x = _normal_width
	$Sprite2D.scale.x = 1.0
	_half_width = _normal_width / 2.0
	_wide_timer = null
```
