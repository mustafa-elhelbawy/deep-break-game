### Step 1 — Create placeholder SFX files

Godot requires real `.ogg` files to import. Generate four minimal silent `.ogg` files using Python so the project imports cleanly:

```bash
mkdir -p assets/audio/sfx
python3 - <<'EOF'
# Writes a minimal valid OGG Vorbis file (44-byte header, 1 channel, 8000 Hz, silent)
import struct, os

def write_silent_ogg(path: str) -> None:
    """
    Writes the smallest legal OGG Vorbis bitstream accepted by most decoders.
    The file contains an identification header, comment header, setup header,
    and one silent audio packet — enough for Godot to import successfully.
    """
    # Minimal OGG page header helper
    def ogg_page(serial: int, seq: int, granule: int, flag: int, data: bytes) -> bytes:
        seg_count = (len(data) + 254) // 255
        segs = []
        remaining = len(data)
        for _ in range(seg_count - 1):
            segs.append(255)
            remaining -= 255
        segs.append(remaining % 256 if remaining != 255 else 255)

        header = (
            b"OggS"
            + struct.pack("<B", 0)           # version
            + struct.pack("<B", flag)        # header type
            + struct.pack("<q", granule)     # granule position
            + struct.pack("<I", serial)      # stream serial
            + struct.pack("<I", seq)         # page sequence
            + struct.pack("<I", 0)           # checksum placeholder
            + struct.pack("<B", len(segs))   # segment count
            + bytes(segs)
        )
        page = header + data

        # CRC-32 (OGG variant, poly 0x04C11DB7)
        crc = 0
        for byte in page:
            crc ^= byte << 24
            for _ in range(8):
                crc = ((crc << 1) ^ 0x04C11DB7) if crc & 0x80000000 else crc << 1
            crc &= 0xFFFFFFFF
        return header[:22] + struct.pack("<I", crc) + header[26:] + data

    # Vorbis identification header (7 + 22 bytes)
    id_header = (
        b"\x01vorbis"
        + struct.pack("<I", 0)       # version
        + struct.pack("<B", 1)       # channels
        + struct.pack("<I", 8000)    # sample rate
        + struct.pack("<i", -1)      # max bitrate
        + struct.pack("<i", 16000)   # nominal bitrate
        + struct.pack("<i", -1)      # min bitrate
        + struct.pack("<B", 0xB8)    # blocksize
        + struct.pack("<B", 1)       # framing
    )
    # Vorbis comment header
    comment_header = (
        b"\x03vorbis"
        + struct.pack("<I", 0)       # vendor string length
        + struct.pack("<I", 0)       # comment count
        + struct.pack("<B", 1)       # framing
    )

    serial = 0x12345678
    with open(path, "wb") as f:
        f.write(ogg_page(serial, 0, 0, 0x02, id_header))
        f.write(ogg_page(serial, 1, 0, 0x00, comment_header))

for name in ["ball_lost", "brick_destroy", "game_over", "level_clear"]:
    write_silent_ogg(f"assets/audio/sfx/{name}.ogg")
    print(f"  wrote assets/audio/sfx/{name}.ogg")
EOF
```

### Step 2 — Overwrite `scripts/autoloads/AudioManager.gd`

```gdscript
# AudioManager.gd
# Autoload singleton — loads OGG files and plays them on demand.
# Uses a fixed pool of AudioStreamPlayer nodes to allow overlapping sounds.
#
# [GDScript] 'extends Node' — autoload attaches to a plain Node in the scene tree.
# [JS equiv]  class AudioManager { play(id) { new Audio(SFX[id]).play() } }
extends Node

# ── Sound pool size ────────────────────────────────────────────────────────────
# How many sounds can play simultaneously without cutting each other off.
# 4 is enough for brick_destroy bursts (≤ 3 bricks per frame in practice).
# [JS equiv]  const POOL_SIZE = 4
const POOL_SIZE: int = 4

# ── Stream map ─────────────────────────────────────────────────────────────────
# Keyed by the string IDs used in AudioManager.play() calls.
# load() at runtime (not preload) so missing files produce a clear error, not a crash.
# [JS equiv]  const SFX = { ball_lost: new Audio('/sfx/ball_lost.ogg'), ... }
var _streams: Dictionary = {}

# ── Player pool ────────────────────────────────────────────────────────────────
# Round-robin index into the player array.
# [JS equiv]  const _pool = Array.from({length: POOL_SIZE}, () => new Audio())
var _players: Array[AudioStreamPlayer] = []
var _pool_idx: int = 0

# ── Lifecycle ──────────────────────────────────────────────────────────────────

func _ready() -> void:
	# Build AudioStreamPlayer pool.
	# [JS equiv]  for (let i = 0; i < POOL_SIZE; i++) _pool.push(new Audio())
	for i in POOL_SIZE:
		var player := AudioStreamPlayer.new()
		add_child(player)
		_players.append(player)

	# Load all known SFX files into the stream map.
	# If a file is missing, load() returns null and play() will safely skip it.
	# [JS equiv]  Object.entries(SFX_PATHS).forEach(([k, v]) => SFX[k] = new Audio(v))
	_load_stream("ball_lost",     "res://assets/audio/sfx/ball_lost.ogg")
	_load_stream("brick_destroy", "res://assets/audio/sfx/brick_destroy.ogg")
	_load_stream("game_over",     "res://assets/audio/sfx/game_over.ogg")
	_load_stream("level_clear",   "res://assets/audio/sfx/level_clear.ogg")

# ── Public API ─────────────────────────────────────────────────────────────────

## Play a sound by its string ID (e.g. "brick_destroy").
## Uses round-robin pool — never blocks; silently skips unknown IDs.
## [JS equiv]  play(id) { SFX[id]?.cloneNode(true).play() }
func play(sound_id: String) -> void:
	if not _streams.has(sound_id):
		return
	var stream: AudioStream = _streams[sound_id]
	if stream == null:
		return

	# Pick the next player in the pool (round-robin).
	# [JS equiv]  const player = _pool[_poolIdx++ % POOL_SIZE]
	var player: AudioStreamPlayer = _players[_pool_idx]
	_pool_idx = (_pool_idx + 1) % POOL_SIZE

	player.stream = stream
	player.play()

# ── Private helpers ────────────────────────────────────────────────────────────

func _load_stream(id: String, path: String) -> void:
	# ResourceLoader.exists() avoids a hard error on missing files.
	# [JS equiv]  try { SFX[id] = new Audio(path) } catch { SFX[id] = null }
	if ResourceLoader.exists(path):
		_streams[id] = load(path)
	else:
		_streams[id] = null
		push_warning("AudioManager: missing SFX file — " + path)
```

### Step 3 — Add paddle-hit angle logic to `Ball.gd`

Find the `_move_and_reflect()` function in `scripts/game/Ball.gd`. Inside the collision loop, after detecting a hit on the paddle (the collider is an `AnimatableBody2D`), replace the raw normal reflection with a deflection based on contact position.

Locate this block (or similar) in `Ball.gd`:

```gdscript
if collision:
    var normal: Vector2 = collision.get_normal()
    _velocity = _velocity.bounce(normal)
```

Replace with:

```gdscript
if collision:
    var normal: Vector2 = collision.get_normal()
    var collider: Object = collision.get_collider()

    # Paddle hit: deflect angle based on where the ball contacts the paddle.
    # [JS equiv]  if (collider.type === 'paddle') { ... }
    if collider is AnimatableBody2D:
        # contact_x is in [-1, +1]: -1 = left edge, 0 = centre, +1 = right edge.
        # [JS equiv]  const t = (ball.x - paddle.x) / (paddle.width / 2)
        var paddle_half: float = collider.get_node("CollisionShape2D").shape.size.x * 0.5
        var contact_x: float = clampf(
            (global_position.x - collider.global_position.x) / paddle_half,
            -1.0, 1.0
        )
        # Map contact_x to a launch angle: max 60° from vertical.
        # [JS equiv]  const angle = t * Math.PI / 3  (60 degrees)
        var angle: float = contact_x * (PI / 3.0)
        _velocity = Vector2(sin(angle), -abs(cos(angle))) * _current_speed
        AudioManager.play("paddle_hit")
    else:
        _velocity = _velocity.bounce(normal)
```

> **Note**: `paddle_hit` SFX is not in the Phase 4 spec — add a silent placeholder `.ogg` and register it in AudioManager alongside the others. Or simply omit the `AudioManager.play("paddle_hit")` line until T038 verifies it; T036 doesn't gate on paddle sound.
