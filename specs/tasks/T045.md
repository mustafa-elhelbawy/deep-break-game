### Step 2 — AnimationPlayer animations

**Animation `"entrance"`** (plays once on _ready, duration 1.2 s):
- Track `Logo:modulate:a`: 0.0 → 1.0 over 0.8 s (fade in logo)
- Track `TapLabel:modulate:a`: 0.0 → 1.0, starts at t=0.6, ends at t=1.2

**Animation `"tap_pulse"`** (loops, duration 1.4 s):
- Track `TapLabel:modulate:a`: 1.0 → 0.3 → 1.0 (pulsing opacity)

**Setup**: in `Title.gd` `_ready()`, play `"entrance"` first; on its completion signal start `"tap_pulse"`.

### Step 3 — Title.gd

Create `scenes/main/Title.gd` and attach to the `Title` root node:

```gdscript
# Title.gd
# Title screen — shows logo, waits for tap, then transitions to the game.
# [GDScript] extends Node2D — this scene owns no physics bodies.
# [JS equiv]  class TitleScreen extends HTMLElement { ... }
extends Node2D

@onready var _anim: AnimationPlayer = $AnimationPlayer
@onready var _tap_label: Label = $TapLabel

# _ready plays the entrance animation. TAP is not active until entrance finishes.
# [JS equiv]  connectedCallback() { this._anim.play('entrance') }
func _ready() -> void:
	# Hide tap label until entrance completes.
	_tap_label.modulate.a = 0.0
	_anim.play("entrance")
	# [GDScript] connect signal with a Callable lambda — like addEventListener with arrow fn.
	# [JS equiv]  anim.addEventListener('animationFinished', (name) => { if(name==='entrance') ... })
	_anim.animation_finished.connect(func(anim_name: String) -> void:
		if anim_name == "entrance":
			_anim.play("tap_pulse")
	)

# _input receives all events. Only accept touches after entrance finishes.
# [GDScript] 'is' is a runtime type check — like 'instanceof' in JS.
# [JS equiv]  document.addEventListener('touchstart', handler)
func _input(event: InputEvent) -> void:
	# Only allow tap if entrance is done (tap_pulse is the active animation).
	# [GDScript] current_animation returns the currently playing animation name string.
	if _anim.current_animation != "tap_pulse":
		return
	if event is InputEventScreenTouch and event.pressed:
		# start_game() transitions GameManager to WAITING — Main.gd will swap to Game.tscn.
		# [JS equiv]  gameManager.startGame()
		GameManager.start_game()
```
