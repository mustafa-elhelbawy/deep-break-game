### Step 1 — LevelGenerator.gd: add collapse constants

Open `scripts/autoloads/LevelGenerator.gd`.

**Add constants** (after the `POWERUP_TABLE` constant added in T051):
```gdscript
# Mobile brick assignment — chance and speed range by level tier.
# Indestructible bricks (hp == 0) CANNOT be mobile — they are fixed obstacles.
# [GDScript] Array of Dictionaries — like JS: [{ threshold: 10, chance: 0.15, ... }, ...]
const MOBILE_TABLE: Array = [
	{ "threshold": 10, "chance": 0.15, "speed_min": 30.0, "speed_max": 50.0 },
	{ "threshold": 20, "chance": 0.20, "speed_min": 50.0, "speed_max": 90.0 },
]
```

**Add helper to get mobile config**:
```gdscript
## Returns the mobile brick config for the given level, or an empty Dictionary if none.
## [JS equiv]  _getMobileConfig(level) { return table.filter(...).at(-1) ?? {} }
func _get_mobile_config(level: int) -> Dictionary:
	var config: Dictionary = {}
	for entry in MOBILE_TABLE:
		if level >= entry["threshold"]:
			config = entry
	return config
```

### Step 2 — Add collapse_speed to brick descriptor loop

Inside `generate()`, after the `powerup_type` assignment (added in T051) and before `result.append()`:

```gdscript
			# Determine collapse speed for this brick.
			# Only destructible bricks (hp > 0) can be mobile — indestructible bricks are anchors.
			# [JS equiv]  const collapseSpeed = hp > 0 && mobile && rng.random() < config.chance ? rand(min, max) : 0
			var collapse_speed: float = 0.0
			if hp > 0:
				var mob_config: Dictionary = _get_mobile_config(level)
				if not mob_config.is_empty() and rng.randf() < mob_config["chance"]:
					# randf_range(min, max) = random float in [min, max].
					# [JS equiv]  Math.random() * (max - min) + min
					collapse_speed = rng.randf_range(mob_config["speed_min"], mob_config["speed_max"])

			result.append({
				"grid_pos":       Vector2i(col, row),
				"hit_points":     hp,
				"tier":           tier,
				"powerup_type":   powerup_type,
				"collapse_speed": collapse_speed,   # 0.0 = static
			})
```

> **RNG order rule**: the `rng.randf()` and `rng.randf_range()` calls for `collapse_speed` are added AFTER the `powerup_type` rng calls. Never insert rng calls between existing ones — it shifts all downstream draws and breaks seed reproducibility.
