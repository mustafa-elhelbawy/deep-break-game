### Step 2 — Write `Main.gd`

```gdscript
# Main.gd
# Scene root — manages transitions between Title and Game scenes.
# Listens to GameManager state changes and swaps the active child scene accordingly.
#
# Scene swap strategy: load the new scene, free the old one.
# Using change_scene_to_file() is avoided because it resets the entire tree including
# autoloads; instead we add/remove child scenes manually to keep GameManager state intact.
#
# [GDScript] extends Node — no 2D or 3D rendering needed at this level.
# [JS equiv]  class AppRouter { ... } — a top-level controller with no visual output.
extends Node

# ── Scene paths ────────────────────────────────────────────────────────────────
# Preloading ensures the packed scenes are ready before the first swap.
# [GDScript] preload() resolves the path at parse time, not at runtime.
# [JS equiv]  import TitleScene from './scenes/main/Title.js'  (static import)
const TitleScene: PackedScene = preload("res://scenes/main/Title.tscn")
const GameScene:  PackedScene = preload("res://scenes/game/Game.tscn")

# ── State ──────────────────────────────────────────────────────────────────────
# [GDScript] Typed variable holding the current active child scene node.
# [JS equiv]  #currentScene = null  (private class field)
var _current_scene: Node = null

# ── Lifecycle ──────────────────────────────────────────────────────────────────

func _ready() -> void:
	# Connect to GameManager state transitions.
	# Every state change anywhere in the game will call _on_state_changed.
	# [JS equiv]  gameManager.addEventListener('stateChanged', this._onStateChanged.bind(this))
	GameManager.state_changed.connect(_on_state_changed)

	# Boot: GameManager starts in TITLE state (set in GameManager._ready, or default).
	# Load the title screen on first run.
	# [JS equiv]  this._loadScene('title')
	_load_title()

# ── Scene loading ──────────────────────────────────────────────────────────────

func _load_title() -> void:
	_swap_to(TitleScene)
	# Ensure GameManager is in TITLE state so Title.gd tap works correctly.
	# If it is already TITLE (first launch), this is a no-op on state.
	# [GDScript] Direct property comparison — no method call needed.
	if GameManager.state != GameManager.State.TITLE:
		GameManager.state = GameManager.State.TITLE
		GameManager.emit_signal("state_changed", GameManager.State.TITLE)

func _load_game() -> void:
	_swap_to(GameScene)

func _swap_to(packed: PackedScene) -> void:
	# Free the current scene before loading the new one.
	# queue_free() is deferred — the node is freed at the end of the current frame.
	# [JS equiv]  if (this._currentScene) this._currentScene.remove()
	if _current_scene != null and is_instance_valid(_current_scene):
		_current_scene.queue_free()

	# Instantiate and add the new scene as a child of Main.
	# [GDScript] PackedScene.instantiate() = new SceneClass() in JS.
	_current_scene = packed.instantiate()
	add_child(_current_scene)

# ── State reactor ──────────────────────────────────────────────────────────────

func _on_state_changed(new_state: GameManager.State) -> void:
	# WAITING coming from TITLE means the player tapped "play" — load the game.
	# All other WAITING transitions (after ball lost, level clear) happen INSIDE
	# Game.tscn and do not require a scene swap here.
	match new_state:
		GameManager.State.WAITING:
			# Only swap to game if we are currently showing the title.
			# [GDScript] 'is' checks node type — like 'instanceof' in JS.
			if _current_scene is Node and _current_scene.scene_file_path == TitleScene.resource_path:
				_load_game()

		GameManager.State.TITLE:
			# Game has ended and requested a return to the title screen.
			# This is triggered in the future by a "Main Menu" button (Phase 7).
			# For now it fires if GameManager.state is set to TITLE externally.
			_load_title()
```

### Step 4b — AMEND: Fix Game.gd._ready() — remove start_game() call

> **Architecture amendment (ADR-005)**: `Game.gd._ready()` currently calls `GameManager.start_game()` directly. When `Main.gd` loads `Game.tscn` AFTER `Title.gd` already called `start_game()` (state is already WAITING), `_ready()` calling it again resets score and lives to zero.

Open `scenes/game/Game.gd`. Find `_ready()`. Replace the direct `GameManager.start_game()` call with a state-reactor call:

**Before** (current code):
```gdscript
func _ready() -> void:
    JuiceManager.register_camera(_camera)
    GameManager.state_changed.connect(_on_state_changed)
    # ... other signal connections ...
    GameManager.start_game()   # ← REMOVE this line
```

**After**:
```gdscript
func _ready() -> void:
    JuiceManager.register_camera(_camera)
    GameManager.state_changed.connect(_on_state_changed)
    # ... other signal connections ...
    # React to the current state — Main.gd loaded us because state is already WAITING.
    # Manually fire the handler so the first level spawns correctly.
    # [JS equiv]  this._onStateChanged(gameManager.state)
    _on_state_changed(GameManager.state)
```

This makes `Game.gd` a pure consumer of `GameManager` state rather than a producer. `start_game()` is now only called from `Title.gd` when the player taps.
