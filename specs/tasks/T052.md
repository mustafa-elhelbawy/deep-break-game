### Step 1 — Brick.gd: store powerup_type and drop on destroy

Open `scenes/game/Brick.gd`.

**Add preload** near the top:
```gdscript
# Preload the PowerUp packed scene — instantiated on destroy if type is set.
# [GDScript] preload() = static import resolved at parse time.
# [JS equiv]  import PowerUpScene from './PowerUp.js'
const PowerUpScene: PackedScene = preload("res://scenes/game/PowerUp.tscn")
```

**Add state variable** in the state section:
```gdscript
# Power-up type this brick will drop on destroy. "" = none.
# [JS equiv]  #powerupType = ''  (private class field)
var _powerup_type: String = ""
```

**Extend the `spawn()` or setup method** that reads the descriptor. Find where `hit_points` and `tier` are read from the descriptor and add:
```gdscript
# Read powerup_type from the descriptor. Default to "" if key absent (backwards-safe).
# [GDScript] Dictionary.get(key, default) — like JS nullish coalescing: dict?.key ?? ''
_powerup_type = descriptor.get("powerup_type", "")
```

**In the destroy path of `take_hit()`** (when `_hit_points <= 0`), add the power-up drop BEFORE calling `queue_free()`:
```gdscript
		# Drop a power-up token if this brick carries one.
		if _powerup_type != "":
			_drop_powerup()
		# ... existing destroy logic (signal emit, queue_free, etc.) ...
```

**Add the `_drop_powerup()` helper method**:
```gdscript
func _drop_powerup() -> void:
	# Instantiate the token and add it to the same parent (BrickGrid or Game scene root).
	# We add to the scene tree root so the token persists after the brick is freed.
	# [JS equiv]  const token = new PowerUpToken(); document.body.appendChild(token)
	var token: Area2D = PowerUpScene.instantiate()
	get_tree().current_scene.add_child(token)
	# Position the token at the brick's world position.
	# [GDScript] global_position = the node's position in world space.
	token.global_position = global_position
	# Call setup() to assign type and colour (defined in PowerUp.gd).
	token.setup(_powerup_type)
	# Connect the caught signal so Game.gd can react.
	# Game.gd is the current scene root — it exposes _on_powerup_caught().
	# [JS equiv]  token.addEventListener('caught', (type) => game._onPowerupCaught(type))
	var game_root: Node = get_tree().current_scene
	if game_root.has_method("_on_powerup_caught"):
		token.caught.connect(game_root._on_powerup_caught)
```
