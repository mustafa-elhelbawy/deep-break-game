### Step 2 — Game.gd: wire flip banner

Open `scenes/game/Game.gd`.

**Add node reference**:
```gdscript
@onready var _flip_banner: CanvasLayer = $FlipBanner
```

**Add state variable**:
```gdscript
# True while the "FLIP LEVEL!" banner is showing — blocks ball launch input.
var _showing_flip_banner: bool = false
```

**Add `_show_flip_banner()` coroutine**:
```gdscript
## Shows the FLIP LEVEL! banner for 2 seconds, then hides it.
## Uses await so ball launch is blocked during display.
## [GDScript] Coroutine — 'await' suspends this function but NOT the game loop.
## [JS equiv]  async _showFlipBanner() { show(); await sleep(2000); hide() }
func _show_flip_banner() -> void:
	_showing_flip_banner = true
	_flip_banner.visible = true

	# Animate: scale from 1.2 → 1.0 (pop-in effect) over 0.2 s.
	var label: Label = _flip_banner.get_node("FlipLabel")
	label.scale = Vector2(1.3, 1.3)
	var tween := create_tween()
	tween.tween_property(label, "scale", Vector2(1.0, 1.0), 0.2)

	# Wait 2 seconds total.
	# [GDScript] await suspends this coroutine — _physics_process keeps running normally.
	# [JS equiv]  await new Promise(resolve => setTimeout(resolve, 2000))
	await get_tree().create_timer(2.0).timeout

	# Fade out.
	tween = create_tween()
	tween.tween_property(label, "modulate:a", 0.0, 0.3)
	await tween.finished

	_flip_banner.visible = false
	label.modulate.a = 1.0   # reset alpha for next use
	_showing_flip_banner = false
```

**Update `_spawn_level()` to show the banner for flipped levels**:
```gdscript
func _spawn_level() -> void:
	var level_data: Dictionary = LevelGenerator.generate(GameManager.seed, GameManager.level)
	var descriptors: Array = level_data["bricks"]
	_flipped = level_data["layout_flipped"]

	_apply_flip(_flipped)
	_grid.spawn(descriptors, _flipped)
	_wire_mobile_bricks()

	# Show FLIP banner for flipped levels — blocks launch until banner clears.
	if _flipped:
		_show_flip_banner()   # async — ball input is blocked via _showing_flip_banner
```

**Update `_input()` to respect the banner block**:
```gdscript
func _input(event: InputEvent) -> void:
	if GameManager.state != GameManager.State.WAITING:
		return
	# Block ball launch while the flip banner is animating.
	# [JS equiv]  if (this._showingFlipBanner) return
	if _showing_flip_banner:
		return
	if event is InputEventScreenTouch and event.pressed:
		_launch_ball()
```
