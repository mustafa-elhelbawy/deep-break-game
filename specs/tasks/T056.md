### Step 2 — HUD.gd: add show/hide methods

Open `scenes/ui/HUD.gd`. Add node references and methods:

```gdscript
@onready var _powerup_container: VBoxContainer = $PowerUpContainer
@onready var _powerup_label: Label             = $PowerUpContainer/PowerUpLabel
@onready var _powerup_bar: ColorRect           = $PowerUpContainer/PowerUpBar

# Type-to-display-label map.
# [GDScript] Dictionary = like a plain JS object.
const POWERUP_LABELS: Dictionary = {
	"W": "W: WIDE",
	"S": "S: SLOW",
	"M": "M: MULTI",
}

# Type-to-colour map — bar colour matches power-up identity.
const POWERUP_COLOURS: Dictionary = {
	"W": Color("#00c8b8"),   # teal
	"S": Color("#00e0ff"),   # cyan
	"M": Color("#ffc107"),   # amber
}

## Show the power-up indicator. duration=0 means no countdown bar (used for M).
## [JS equiv]  showPowerup(type, duration) { ... }
func show_powerup(type: String, duration: float) -> void:
	_powerup_label.text   = POWERUP_LABELS.get(type, type)
	_powerup_bar.color    = POWERUP_COLOURS.get(type, Color.WHITE)
	_powerup_container.visible = true

	if duration > 0.0:
		# Animate bar width from 100 to 0 over the duration.
		# create_tween() returns a fire-and-forget tween.
		# [JS equiv]  element.animate([{width:'100px'},{width:'0px'}], {duration: duration*1000})
		var tween := create_tween()
		tween.tween_property(_powerup_bar, "size:x", 0.0, duration)
		tween.tween_callback(hide_powerup)   # auto-hide when bar empties
	else:
		# M: no countdown — bar stays full width (no tween).
		_powerup_bar.size.x = 100.0

## Hide the power-up indicator. Called on expiry or when all balls are lost.
## [JS equiv]  hidePowerup() { this._container.style.display = 'none' }
func hide_powerup() -> void:
	_powerup_container.visible = false
	_powerup_label.text = ""
	_powerup_bar.size.x = 100.0  # reset bar for next use
```

### Step 3 — Game.gd: wire show_powerup calls

Open `scenes/game/Game.gd`. Replace the `# TODO T056` comments added in T053–T055:

**In `_activate_wide()` → after `_paddle.activate_wide()`**:
```gdscript
_hud.show_powerup("W", Paddle.WIDE_DURATION)
```

**In `_activate_slow()` → after applying slow**:
```gdscript
_hud.show_powerup("S", SLOW_DURATION)
```

**In `_activate_multiball()` → after spawning extra ball**:
```gdscript
_hud.show_powerup("M", 0.0)
```

**In `_clear_extra_balls()` → after clearing**:
```gdscript
_hud.hide_powerup()
```

**In `_revert_slow()` → after reverting** (in case tween callback doesn't fire, belt-and-suspenders):
```gdscript
# HUD bar auto-hides via tween callback; explicit call here is belt-and-suspenders.
# No harm if called twice — hide_powerup() is idempotent.
_hud.hide_powerup()
```

> **Note**: `Paddle.WIDE_DURATION` — since it is a const on `Paddle`, you may need to access it as `_paddle.WIDE_DURATION` or define a matching const in `Game.gd`. The simplest approach is to define `const WIDE_DURATION: float = 10.0` in `Game.gd` matching the value in `Paddle.gd`.
